cmake_minimum_required(VERSION 3.16)
project(OpimizedMW_GL_P2_Initial_QC LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
option(OPIMIZEDMW_P2_SANITIZERS "Address and undefined-behavior sanitizers" OFF)
option(OPIMIZEDMW_P2_OSG_TESTS "Compare production adapter against real OSG" ON)
enable_testing()
add_executable(opimizedmw-p2-vertex-tests vertex-tests.cpp)
set(targets opimizedmw-p2-vertex-tests)
if(OPIMIZEDMW_P2_OSG_TESTS)
    find_package(OpenSceneGraph 3.6 REQUIRED COMPONENTS osg osgDB osgUtil)
    add_executable(opimizedmw-p2-osg-tests osg-tests.cpp "${ROOT}/components/sceneutil/pagingwork.cpp")
    target_include_directories(opimizedmw-p2-osg-tests SYSTEM PRIVATE ${OPENSCENEGRAPH_INCLUDE_DIRS})
    target_link_libraries(opimizedmw-p2-osg-tests PRIVATE ${OPENSCENEGRAPH_LIBRARIES})
    list(APPEND targets opimizedmw-p2-osg-tests)
endif()
foreach(t IN LISTS targets)
    target_include_directories(${t} PRIVATE "${ROOT}")
    target_link_libraries(${t} PRIVATE Threads::Threads)
    if(MSVC)
        target_compile_options(${t} PRIVATE /W4 /WX /permissive- /EHsc)
    else()
        target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        if(OPIMIZEDMW_P2_SANITIZERS)
            target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${t} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
    add_test(NAME ${t} COMMAND ${t})
    set_tests_properties(${t} PROPERTIES TIMEOUT 180)
endforeach()
