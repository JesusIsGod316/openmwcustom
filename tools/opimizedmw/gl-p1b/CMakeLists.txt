cmake_minimum_required(VERSION 3.16)
project(OpimizedMW_GL_P1B_QC LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
option(OPIMIZEDMW_P1B_SANITIZERS "Address and undefined-behavior sanitizers" OFF)
option(OPIMIZEDMW_P1B_OSG_TESTS "Compile real OSG cache and lifetime tests" ON)
enable_testing()
add_executable(opimizedmw-budget-tests budget-tests.cpp "${ROOT}/components/misc/hostmemory.cpp")
add_executable(opimizedmw-pressure-tests "${ROOT}/tools/opimizedmw/gl-p1a/pressure-tests.cpp" "${ROOT}/components/misc/hostmemory.cpp")
add_executable(legacy-host-memory-tests "${ROOT}/tools/v4/cp4/host-memory-tests.cpp" "${ROOT}/components/misc/hostmemory.cpp")
set(targets opimizedmw-budget-tests opimizedmw-pressure-tests legacy-host-memory-tests)
if(OPIMIZEDMW_P1B_OSG_TESTS)
    find_package(OpenSceneGraph 3.6 REQUIRED COMPONENTS osg osgDB osgUtil)
    add_executable(opimizedmw-cache-tests cache-tests.cpp "${ROOT}/components/resource/multiobjectcache.cpp")
    add_executable(legacy-cache-tests "${ROOT}/tools/v4/cp4/cache-pressure-tests.cpp" "${ROOT}/components/resource/multiobjectcache.cpp")
    foreach(t opimizedmw-cache-tests legacy-cache-tests)
        target_include_directories(${t} SYSTEM PRIVATE ${OPENSCENEGRAPH_INCLUDE_DIRS})
        target_link_libraries(${t} PRIVATE ${OPENSCENEGRAPH_LIBRARIES})
    endforeach()
    list(APPEND targets opimizedmw-cache-tests legacy-cache-tests)
endif()
foreach(t IN LISTS targets)
    target_include_directories(${t} PRIVATE "${ROOT}")
    target_link_libraries(${t} PRIVATE Threads::Threads)
    if(MSVC)
        target_compile_options(${t} PRIVATE /W4 /WX /permissive- /EHsc)
        target_link_libraries(${t} PRIVATE psapi)
    else()
        target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
            # Existing diagnostics intentionally evaluates every copy call.
            target_compile_options(${t} PRIVATE -Wno-bitwise-instead-of-logical)
        endif()
        if(OPIMIZEDMW_P1B_SANITIZERS)
            target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${t} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
    add_test(NAME ${t} COMMAND ${t})
    set_tests_properties(${t} PROPERTIES TIMEOUT 60)
endforeach()
