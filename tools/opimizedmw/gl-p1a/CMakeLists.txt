cmake_minimum_required(VERSION 3.16)
project(OpimizedMW_GL_P1A_QC LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
option(OPIMIZEDMW_P1A_SANITIZERS "Run address and undefined-behavior sanitizers" OFF)
enable_testing()
add_executable(opimizedmw-pressure-tests pressure-tests.cpp "${ROOT}/components/misc/hostmemory.cpp")
add_executable(legacy-host-memory-tests "${ROOT}/tools/v4/cp4/host-memory-tests.cpp"
    "${ROOT}/components/misc/hostmemory.cpp")
foreach(target opimizedmw-pressure-tests legacy-host-memory-tests)
    target_include_directories(${target} PRIVATE "${ROOT}")
    target_link_libraries(${target} PRIVATE Threads::Threads)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
        target_link_libraries(${target} PRIVATE psapi)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        if(OPIMIZEDMW_P1A_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
    add_test(NAME ${target} COMMAND ${target})
    set_tests_properties(${target} PROPERTIES TIMEOUT 60)
endforeach()
