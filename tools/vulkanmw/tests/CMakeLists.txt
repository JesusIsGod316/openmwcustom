cmake_minimum_required(VERSION 3.20)
project(VulkanMWPhase3CMetadataTests LANGUAGES CXX)

find_package(glm CONFIG REQUIRED)
enable_testing()
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME vulkanmw-isolated-profile
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test-vulkan-profile.py)
add_test(NAME vulkanmw-frame-profile-report
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test-frame-profile-report.py)
add_test(NAME vulkanmw-phase3c-benchmark-control
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test-benchmark-control.py)
add_test(NAME vulkanmw-producer-launcher
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test-producer-launcher.py)
add_test(NAME vulkanmw-publication-cohort
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test-publication-cohort.py)
add_executable(vulkanmw-phase3c-metadata-tests phase3c-metadata-tests.cpp)
target_compile_features(vulkanmw-phase3c-metadata-tests PRIVATE cxx_std_20)
target_include_directories(vulkanmw-phase3c-metadata-tests PRIVATE ../../..)
target_link_libraries(vulkanmw-phase3c-metadata-tests PRIVATE glm::glm)
if(MSVC)
    target_compile_options(vulkanmw-phase3c-metadata-tests PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(vulkanmw-phase3c-metadata-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME vulkanmw-phase3c-metadata COMMAND vulkanmw-phase3c-metadata-tests)

foreach(name IN ITEMS population-delta actor-preparation skeletal-object frame-profile group-publication fx-interface frame-resource-pool change-driven producer-queue)
    if(name STREQUAL "actor-preparation")
        set(source "../../v4/cp4/actor-preparation-tests.cpp")
    elseif(name STREQUAL "skeletal-object")
        set(source "skeletal-object-tests.cpp")
    elseif(name STREQUAL "frame-profile")
        set(source "frame-profile-tests.cpp")
    elseif(name STREQUAL "group-publication")
        set(source "group-publication-tests.cpp")
    elseif(name STREQUAL "fx-interface")
        set(source "fx-interface-tests.cpp")
    elseif(name STREQUAL "frame-resource-pool")
        set(source "frame-resource-pool-tests.cpp")
    elseif(name STREQUAL "change-driven")
        set(source "change-driven-tests.cpp")
    elseif(name STREQUAL "producer-queue")
        set(source "producer-queue-tests.cpp")
    else()
        set(source "population-delta-tests.cpp")
    endif()
    add_executable(vulkanmw-${name}-tests ${source})
    target_compile_features(vulkanmw-${name}-tests PRIVATE cxx_std_20)
    target_include_directories(vulkanmw-${name}-tests PRIVATE ../../..)
    target_link_libraries(vulkanmw-${name}-tests PRIVATE glm::glm Threads::Threads)
    if(MSVC)
        target_compile_definitions(vulkanmw-${name}-tests PRIVATE _CRT_SECURE_NO_WARNINGS)
        target_compile_options(vulkanmw-${name}-tests PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(vulkanmw-${name}-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME vulkanmw-${name} COMMAND vulkanmw-${name}-tests)
endforeach()

# Existing baseline contracts must still pass after changing the producer and
# publisher. These fixtures are renderer-data-only and need no game window.
foreach(name IN ITEMS static-population-smoke static-sync-state-tests publication-runtime-tests)
    add_executable(vulkanmw-${name} ../../v4/cp4/${name}.cpp)
    target_compile_features(vulkanmw-${name} PRIVATE cxx_std_20)
    target_include_directories(vulkanmw-${name} PRIVATE ../../..)
    target_link_libraries(vulkanmw-${name} PRIVATE glm::glm)
    if(MSVC)
        target_compile_definitions(vulkanmw-${name} PRIVATE _CRT_SECURE_NO_WARNINGS)
        target_compile_options(vulkanmw-${name} PRIVATE /UNDEBUG)
    else()
        target_compile_options(vulkanmw-${name} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME vulkanmw-${name} COMMAND vulkanmw-${name})
endforeach()
add_test(NAME vulkanmw-publication-full-control COMMAND vulkanmw-publication-runtime-tests)
set_tests_properties(vulkanmw-publication-full-control PROPERTIES ENVIRONMENT "OPENMW_V4_FULL_WORLD_PUBLICATION=1")
