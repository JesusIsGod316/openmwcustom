cmake_minimum_required(VERSION 3.16)
project(OpenMWDiagnosticHeaderTests LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)
option(OPENMW_DIAGNOSTIC_TEST_REAL_QT "Test the real Qt6 keyword macros in both include orders" OFF)
get_filename_component(OPENMW_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
if(OPENMW_DIAGNOSTIC_TEST_REAL_QT)
    find_package(Qt6 REQUIRED COMPONENTS Core)
endif()

add_library(process-probe STATIC
    "${OPENMW_ROOT}/components/debug/runtimeprocessmemory.cpp"
    "${OPENMW_ROOT}/components/misc/hostmemory.cpp")
target_include_directories(process-probe PUBLIC "${OPENMW_ROOT}")
target_compile_features(process-probe PUBLIC cxx_std_20)
target_link_libraries(process-probe PUBLIC Threads::Threads)
if(WIN32)
    target_link_libraries(process-probe PUBLIC psapi)
endif()
function(strict_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endfunction()
strict_warnings(process-probe)

set(variants clean keywords-before keywords-after)
if(OPENMW_DIAGNOSTIC_TEST_REAL_QT)
    list(APPEND variants qt-before qt-after)
endif()
foreach(variant IN LISTS variants)
    set(target "diagnostic-headers-${variant}")
    add_executable(${target} header-consumer.cpp)
    target_link_libraries(${target} PRIVATE process-probe)
    strict_warnings(${target})
    if(variant MATCHES "^(keywords|qt)-")
        target_compile_definitions(${target} PRIVATE OPENMW_TEST_KEYWORDS=1)
    endif()
    if(variant MATCHES "-after$")
        target_compile_definitions(${target} PRIVATE OPENMW_TEST_KEYWORDS_AFTER=1)
    endif()
    if(variant MATCHES "^qt-")
        target_link_libraries(${target} PRIVATE Qt6::Core)
        target_compile_definitions(${target} PRIVATE OPENMW_TEST_REAL_QT=1)
    endif()
    add_test(NAME ${target} COMMAND ${target})
    set_tests_properties(${target} PROPERTIES ENVIRONMENT
        "OPENMW_RUNTIME_DIAGNOSTICS=off;OPENMW_GAMEPLAY_DIAGNOSTICS=0")
endforeach()

# Compile and execute the real platform probe and budget policy before the full
# engine build. No OSG, Vulkan or game data is required for this target.
add_executable(host-memory-policy "${OPENMW_ROOT}/tools/v4/cp4/host-memory-tests.cpp")
target_link_libraries(host-memory-policy PRIVATE process-probe)
strict_warnings(host-memory-policy)
add_test(NAME host-memory-policy COMMAND host-memory-policy)
