cmake_minimum_required(VERSION 3.16)
project(OptimizedMW_P8G4_QC LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(OpenSceneGraph 3.6 REQUIRED COMPONENTS osg osgUtil osgDB osgViewer osgGA)
find_package(OpenGL REQUIRED)
find_package(Threads REQUIRED)
find_path(MYGUI_INCLUDE_DIR MyGUI_Colour.h PATH_SUFFIXES MYGUI REQUIRED)
enable_testing()
option(P8G4_SANITIZERS "Address/UB sanitizer checks" OFF)
set(P8G3_SANITIZERS ${P8G4_SANITIZERS} CACHE BOOL "Retained sanitizers" FORCE)
add_subdirectory(../gl-p8g3 retained-p8g3)
add_executable(p8g4-policy policy-tests.cpp)
add_executable(p8g4-osg osg-tests.cpp "${ROOT}/components/sceneutil/shadowproxygroup.cpp"
    "${ROOT}/components/sceneutil/material.cpp")
add_library(p8g4-production OBJECT "${ROOT}/apps/openmw/mwrender/objectpaging.cpp"
    "${ROOT}/components/sceneutil/mwshadowtechnique.cpp")
foreach(t p8g4-policy p8g4-osg p8g4-production)
    target_include_directories(${t} PRIVATE "${ROOT}")
    target_include_directories(${t} SYSTEM PRIVATE ${OPENSCENEGRAPH_INCLUDE_DIRS} "${MYGUI_INCLUDE_DIR}" /usr/include/bullet)
    target_link_libraries(${t} PRIVATE ${OPENSCENEGRAPH_LIBRARIES} OpenGL::GL Threads::Threads)
    if(MSVC)
        target_compile_options(${t} PRIVATE /W4 /WX /EHsc)
    else()
        target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-unused-parameter -Wno-error=deprecated-declarations)
        if(t STREQUAL "p8g4-production")
            # Existing RenderMutationSource copy-constructor warning, outside this OpenGL change.
            target_compile_options(${t} PRIVATE -Wno-error=extra)
        endif()
        if(P8G4_SANITIZERS AND NOT t STREQUAL "p8g4-production")
            target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${t} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endforeach()
add_test(NAME p8g4-policy-capture COMMAND p8g4-policy "${CMAKE_BINARY_DIR}/capture-test")
add_test(NAME p8g4-native-osg COMMAND p8g4-osg)
set_tests_properties(p8g4-policy-capture p8g4-native-osg PROPERTIES TIMEOUT 90)
