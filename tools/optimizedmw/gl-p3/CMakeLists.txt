cmake_minimum_required(VERSION 3.16)
project(OptimizedMW_GL_P3_QC LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
option(OPTIMIZEDMW_P3_SANITIZERS "Address and undefined-behavior sanitizers" OFF)
find_package(OpenSceneGraph 3.6 REQUIRED COMPONENTS osg osgDB osgUtil)
enable_testing()
add_executable(optimizedmw-p3-submission-tests
    submission-tests.cpp
    "${ROOT}/components/sceneutil/optimizer.cpp"
    "${ROOT}/components/sceneutil/pagingwork.cpp")
target_include_directories(optimizedmw-p3-submission-tests PRIVATE "${ROOT}")
target_include_directories(optimizedmw-p3-submission-tests SYSTEM PRIVATE ${OPENSCENEGRAPH_INCLUDE_DIRS})
target_link_libraries(optimizedmw-p3-submission-tests PRIVATE ${OPENSCENEGRAPH_LIBRARIES})
if(MSVC)
    target_compile_options(optimizedmw-p3-submission-tests PRIVATE /W4 /WX /permissive- /EHsc)
else()
    target_compile_options(optimizedmw-p3-submission-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-unused-parameter)
    if(OPTIMIZEDMW_P3_SANITIZERS)
        target_compile_options(optimizedmw-p3-submission-tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(optimizedmw-p3-submission-tests PRIVATE -fsanitize=address,undefined)
    endif()
endif()
add_test(NAME optimizedmw-p3-submission-tests COMMAND optimizedmw-p3-submission-tests)
set_tests_properties(optimizedmw-p3-submission-tests PROPERTIES TIMEOUT 60)
