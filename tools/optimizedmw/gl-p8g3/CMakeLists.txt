cmake_minimum_required(VERSION 3.16)
project(OptimizedMW_P8G3_QC LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(OpenSceneGraph 3.6 REQUIRED COMPONENTS osg osgUtil osgDB osgViewer osgGA)
find_package(OpenGL REQUIRED)
find_package(Threads REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_path(MYGUI_INCLUDE_DIR MyGUI_Colour.h PATH_SUFFIXES MYGUI REQUIRED)
enable_testing()
option(P8G3_SANITIZERS "Address/UB sanitizer checks" OFF)
add_executable(p8g3-policy policy-tests.cpp)
add_executable(p8g3-batches batch-tests.cpp)
add_executable(p8g3-shaders shader-tests.cpp "${ROOT}/components/shader/shadermanager.cpp"
    "${ROOT}/components/debug/debuglog.cpp" "${ROOT}/components/files/conversion.cpp")
add_library(p8g3-production OBJECT "${ROOT}/apps/openmw/mwrender/groundcover.cpp")
foreach(t p8g3-policy p8g3-batches p8g3-shaders p8g3-production)
    target_include_directories(${t} PRIVATE "${ROOT}")
    target_include_directories(${t} SYSTEM PRIVATE ${OPENSCENEGRAPH_INCLUDE_DIRS} "${MYGUI_INCLUDE_DIR}" /usr/include/bullet)
    target_link_libraries(${t} PRIVATE ${OPENSCENEGRAPH_LIBRARIES} OpenGL::GL Threads::Threads)
    if(MSVC)
        target_compile_options(${t} PRIVATE /W4 /WX /EHsc)
    else()
        target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-unused-parameter)
        if(t STREQUAL "p8g3-shaders" OR t STREQUAL "p8g3-production")
            # Unchanged V3 diagnostic headers use deprecated C++20 u8path. Keep
            # these legacy warnings visible; new policy/batch headers stay -Werror.
            target_compile_options(${t} PRIVATE -Wno-error=deprecated-declarations)
        endif()
        if(P8G3_SANITIZERS AND NOT t STREQUAL "p8g3-shaders" AND NOT t STREQUAL "p8g3-production")
            target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${t} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endforeach()
add_test(NAME policy COMMAND p8g3-policy)
add_test(NAME native-osg COMMAND p8g3-batches)
find_program(XVFB_RUN xvfb-run)
if(UNIX AND NOT XVFB_RUN)
    message(FATAL_ERROR "xvfb-run is required for the production shader gate")
endif()
add_test(NAME actual-deployment COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/deployment-tests.py" "${CMAKE_BINARY_DIR}/deployment")
set_tests_properties(actual-deployment PROPERTIES FIXTURES_SETUP deployed-shaders TIMEOUT 90)
if(XVFB_RUN)
    add_test(NAME deployed-pbr-shader-matrix COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:p8g3-shaders> "${CMAKE_BINARY_DIR}/deployment/staged/resources/shaders" "${CMAKE_BINARY_DIR}/deployed-shader-sources")
    set_tests_properties(deployed-pbr-shader-matrix PROPERTIES FIXTURES_REQUIRED deployed-shaders ENVIRONMENT "LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 240)
    add_test(NAME production-shader-matrix COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:p8g3-shaders> "${ROOT}/files/shaders" "${CMAKE_BINARY_DIR}/shader-sources")
    set_tests_properties(production-shader-matrix PROPERTIES ENVIRONMENT "LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 180)
endif()
set_tests_properties(policy native-osg PROPERTIES TIMEOUT 90)
