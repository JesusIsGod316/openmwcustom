cmake_minimum_required(VERSION 3.20)
project(OptimizedMWTemporalCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
option(P8G5_SANITIZERS "Address/undefined sanitizers" OFF)
find_package(glm REQUIRED)
if(MSVC)
  add_compile_options(/W4 /EHsc)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
if(P8G5_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
include_directories(${ROOT})
enable_testing()
add_executable(p8g5-temporal temporal-tests.cpp)
add_executable(p8g5-contract contract-tests.cpp)
foreach(t p8g5-temporal p8g5-contract)
  target_link_libraries(${t} PRIVATE glm::glm)
  add_test(NAME ${t} COMMAND ${t})
  set_tests_properties(${t} PROPERTIES TIMEOUT 20)
endforeach()
find_package(OpenSceneGraph 3.6 REQUIRED COMPONENTS osg osgUtil osgViewer)
find_package(OpenGL REQUIRED)
add_executable(p8g5-gpu-motion gpu-motion-tests.cpp)
target_include_directories(p8g5-gpu-motion PRIVATE ${OPENSCENEGRAPH_INCLUDE_DIRS})
target_compile_definitions(p8g5-gpu-motion PRIVATE P8G5_SHADER="${ROOT}/files/shaders/compatibility/temporal_camera_motion.frag")
target_link_libraries(p8g5-gpu-motion PRIVATE glm::glm ${OPENSCENEGRAPH_LIBRARIES} OpenGL::GL)
add_test(NAME p8g5-gpu-motion COMMAND p8g5-gpu-motion)
set_tests_properties(p8g5-gpu-motion PROPERTIES TIMEOUT 20)
